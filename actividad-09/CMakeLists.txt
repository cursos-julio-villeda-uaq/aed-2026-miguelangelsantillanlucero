cmake_minimum_required(VERSION 3.16)
project(actividad_09 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_library(suma_regresiva src/suma_regresiva.cpp)
target_include_directories(suma_regresiva PUBLIC include)
add_executable(actividad_09 src/main.cpp)
target_link_libraries(actividad_09 PRIVATE suma_regresiva)
add_executable(test_suma_regresiva tests/test_suma_regresiva.cpp)
target_link_libraries(test_suma_regresiva PRIVATE suma_regresiva)
add_test(NAME suma_regresiva_visible COMMAND test_suma_regresiva)
